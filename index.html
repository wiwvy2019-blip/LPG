/** * ฟังก์ชันคำนวณทางชลศาสตร์ (Henzen-GGA Newton-Raphson Method) * นำฟังก์ชันนี้ไปวางแทนที่ฟังก์ชันคำนวณเดิมในไฟล์ของคุณได้เลย * * @param {Array} sysNodes - Array ของข้อมูลจุดต่อและแหล่งน้ำ * @param {Array} sysLinks - Array ของข้อมูลเส้นท่อ * @returns {Boolean} - คืนค่า true หากคำนวณสำเร็จ */ function solveHydraulics(sysNodes, sysLinks) { if (!sysNodes || sysNodes.length === 0 || !sysLinks || sysLinks.length === 0) { alert("ไม่มีข้อมูลเครือข่ายท่อให้คำนวณ"); return false; } const HW_EXP = 1.852; const HW_COEFF = 10.67; const MAX_ITER = 2000; const TOLERANCE = 0.0001; // ความคลาดเคลื่อนที่ยอมรับได้ (m3/s) // 1. เตรียมข้อมูลและดักจับชื่อตัวแปรให้เข้ากับโค้ดดั้งเดิม let nodeMap = {}; let hasReservoir = false; sysNodes.forEach(n => { // รองรับชื่อ Property หลายรูปแบบ let elev = parseFloat(n.elevation ?? n.z) || 0; let dem_Lps = parseFloat(n.demand ?? n.q) || 0; let nType = (n.type || '').toLowerCase(); // ตรวจสอบว่ามีแหล่งน้ำหรือไม่ if (nType === 'reservoir' || nType === 'tank' || n.isReservoir) { hasReservoir = true; nType = 'reservoir'; } else { nType = 'node'; } nodeMap[n.id] = { id: n.id, type: nType, elevation: elev, demand: dem_Lps / 1000, // แปลง L/s เป็น m3/s สำหรับคำนวณ // ค่าตั้งต้น: ถ้าเป็นแหล่งน้ำให้ Head = ระดับความสูง, ถ้าจุดต่อทั่วไปให้เดาว่าสูงกว่าระดับดิน 20m head: nType === 'reservoir' ? elev : (elev + 20), sumQ: 0, sumdQdH: 0 }; }); if (!hasReservoir) { alert("ข้อผิดพลาดทางกลศาสตร์: ระบบต้องมี 'แหล่งน้ำ (Reservoir)' อย่างน้อย 1 จุดเพื่อเป็นจุดอ้างอิงพลังงาน (Boundary Condition)"); return false; } let linkData = sysLinks.map(l => { // รองรับชื่อ Property หลายรูปแบบ (Source/Target หรือ n1/n2) let s = l.source ?? l.n1; let t = l.target ?? l.n2; let D_mm = parseFloat(l.diameter ?? l.d) || 100; let L_m = parseFloat(l.length ?? l.l) || 10; let C_val = parseFloat(l.c ?? l.c_value) || 100; let D_m = D_mm / 1000; // แปลง mm เป็น m // Resistance Factor (R) ในสมการ Hazen-Williams // hf = R * Q * |Q|^(0.852) let R = (HW_COEFF * L_m) / (Math.pow(C_val, HW_EXP) * Math.pow(D_m, 4.87)); return { id: l.id, source: s, target: t, R: R, area: Math.PI * Math.pow(D_m / 2, 2), flow: 0 }; }); // 2. เริ่มลูปคำนวณแบบ Newton-Raphson let iter = 0; let maxError = 1; while (iter < MAX_ITER && maxError > TOLERANCE) { maxError = 0; // ล้างค่าสะสมที่จุดต่อ Object.values(nodeMap).forEach(n => { n.sumQ = 0; n.sumdQdH = 0; }); // คำนวณอัตราการไหลในท่อแต่ละเส้นตามค่า Head ปัจจุบัน linkData.forEach(link => { let n1 = nodeMap[link.source]; let n2 = nodeMap[link.target]; if (!n1 || !n2) return; let dH = n1.head - n2.head; let absDh = Math.abs(dH); let dir = Math.sign(dH); let Q = 0; let dQdH = 0; if (absDh > 1e-6) { // Q = (dH / R)^(1/1.852) Q = dir * Math.pow(absDh / link.R, 1 / HW_EXP); // อนุพันธ์ dQ/dH dQdH = (1 / HW_EXP) * Math.pow(1 / link.R, 1 / HW_EXP) * Math.pow(absDh, (1 / HW_EXP) - 1); } link.flow = Q; // น้ำไหลเข้าเป็นบวก ไหลออกเป็นลบ n1.sumQ -= Q; n2.sumQ += Q; n1.sumdQdH += dQdH; n2.sumdQdH += dQdH; }); // ปรับแก้ค่า Head ที่จุดต่อ Object.values(nodeMap).forEach(n => { if (n.type === 'reservoir') return; // แหล่งน้ำค่า Head คงที่ let netFlow = n.sumQ - n.demand; let errorMagnitude = Math.abs(netFlow); if (errorMagnitude > maxError) maxError = errorMagnitude; if (n.sumdQdH > 1e-10) { let deltaHead = netFlow / n.sumdQdH; n.head += deltaHead * 0.6; // 0.6 คือ Relaxation factor ป้องกันการแกว่ง } }); iter++; } // 3. นำผลลัพธ์ที่สมดุลแล้ว อัปเดตกลับไปยัง Object เดิมของคุณ sysNodes.forEach(n => { let calcData = nodeMap[n.id]; if (calcData) { n.head = parseFloat(calcData.head.toFixed(3)); n.pressure = parseFloat((calcData.head - calcData.elevation).toFixed(3)); } }); sysLinks.forEach(l => { let calcData = linkData.find(x => x.id === l.id); if (calcData) { let s = l.source ?? l.n1; let t = l.target ?? l.n2; // อัปเดตค่า Flow (L/s) l.flow = parseFloat((calcData.flow * 1000).toFixed(3)); // อัปเดตค่าความเร็ว Velocity (m/s) l.velocity = parseFloat((Math.abs(calcData.flow) / calcData.area).toFixed(3)); // รองรับการบันทึกกลับไปที่ตัวแปร vel ตามโค้ดบางเวอร์ชัน l.vel = l.velocity; let h1 = nodeMap[s].head; let h2 = nodeMap[t].head; // อัปเดตค่า Headloss (m) l.headloss = parseFloat(Math.abs(h1 - h2).toFixed(3)); l.hl = l.headloss; } }); console.log(`คำนวณสำเร็จใน ${iter} รอบ (Error สูงสุด: ${(maxError*1000).toFixed(4)} L/s)`); return true; }
